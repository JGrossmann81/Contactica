# Contactica: a small CRM for finding and winning new music-retail accounts.
cmake_minimum_required(VERSION 3.24)
project(Contactica VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()

add_executable(contactica_coretest src/test/coretest.cpp)
target_include_directories(contactica_coretest PRIVATE src third_party)
if(MSVC)
  target_compile_options(contactica_coretest PRIVATE /W3 /EHsc /utf-8)
else()
  target_compile_options(contactica_coretest PRIVATE -Wall -O2)
endif()

include(FetchContent)
set(CONTACTICA_IMGUI_DIR "" CACHE PATH "Use an existing Dear ImGui checkout (v1.93.0-docking) instead of downloading one")
if(CONTACTICA_IMGUI_DIR)
  set(imgui_SOURCE_DIR ${CONTACTICA_IMGUI_DIR})
else()
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.93.0-docking GIT_SHALLOW TRUE SOURCE_SUBDIR _sources_only)
  FetchContent_MakeAvailable(imgui)
endif()
set(IMGUI_SRC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp)
if(WIN32)
  list(APPEND IMGUI_SRC ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
endif()
add_library(contactica_imgui STATIC ${IMGUI_SRC})
target_include_directories(contactica_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/misc/cpp ${imgui_SOURCE_DIR}/backends)
if(WIN32)
  target_link_libraries(contactica_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
endif()
add_library(contactica_app STATIC src/app/App.cpp src/app/Views.cpp)
target_include_directories(contactica_app PUBLIC src third_party)
target_link_libraries(contactica_app PUBLIC contactica_imgui)
if(MSVC)
  target_compile_options(contactica_app PRIVATE /W3 /EHsc /utf-8 /bigobj /MP)
else()
  target_compile_options(contactica_app PRIVATE -Wall -O2)
endif()
add_executable(contactica_uitest src/test/uitest.cpp)
target_link_libraries(contactica_uitest PRIVATE contactica_app)

if(WIN32)
  add_executable(Contactica WIN32 src/app/win/WinMain.cpp res/contactica.rc)
  target_compile_definitions(Contactica PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(Contactica PRIVATE contactica_app shell32 comdlg32 ole32 oleaut32 uuid imm32)
  if(MSVC)
    target_compile_options(Contactica PRIVATE /W3 /EHsc /utf-8)
  else()
    target_link_options(Contactica PRIVATE -static -municode)
  endif()
endif()
